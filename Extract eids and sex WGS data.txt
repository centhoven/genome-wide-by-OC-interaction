#!/bin/bash

# This script copies the .psam file to use eids of participants with WGS data available and their genetic sex for filtering the cohort (in Rstudio)

project="project-GyK0Y6jJ03qkb5JQBF041Yfx"
wgs_file_dir="/Bulk/DRAGEN WGS/DRAGEN population level WGS variants, PLINK format [500k release]/"
data_field="ukb24308"
txt_file_dir="${project}:/Cohort/"

run_copy_psam="cp ${data_field}_c22_b0_v1.psam wgs_eid.psam"

dx run swiss-army-knife \
  -iin="${wgs_file_dir}/${data_field}_c22_b0_v1.psam" \
  -icmd="${run_copy_psam}" \
  --destination="${txt_file_dir}" \
  --instance-type "mem1_ssd1_v2_x4" \
  --priority high \
  --brief --yes
