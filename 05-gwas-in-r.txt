# This script uses the files resulting from '04-pruning' to run GWDIS and GWAS

# First for snp1-2000 of chr 1
#!/bin/bash

project="project-GyK0Y6jJ03qkb5JQBF041Yfx"
data_file_dir="${project}:/Data/"
txt_file_dir="${project}:/Cohort/"

dx run swiss-army-knife \
  -iin="${data_file_dir}/run_GWDIS_mclapply.r" \
  -iin="${data_file_dir}/chr1_251031_pruned.bed" \
  -iin="${data_file_dir}/chr1_251031_pruned.fam" \
  -iin="${txt_file_dir}/pheno_final_imputed_cox.csv" \
  -icmd="export SNP_START=1;
         export SNP_END=2000;
         export BEDFILE=/mnt/project/Data/chr1_251031_pruned.bed;
         export PHENOFILE=/mnt/project/Cohort/pheno_final_imputed_cox.csv;
         Rscript run_GWDIS_mclapply.r" \
  --destination "${data_file_dir}" \
  --instance-type "mem3_ssd1_v2_x8" \
  --priority high \
  --brief --yes

# Then for snp 2001-39000 of chr 1
#!/bin/bash

project="project-GyK0Y6jJ03qkb5JQBF041Yfx"
data_file_dir="${project}:/Data/"
txt_file_dir="${project}:/Cohort/"

# SNP ranges: 2001 to 39000 in batches of 2000
for start in $(seq 2001 2000 39001); do
    end=$(( start + 1999 ))
    if [ $end -gt 39000 ]; then
        end=39000
    fi

    echo "Submitting chr1: SNPs ${start}-${end}"

    dx run swiss-army-knife \
      -iin="${data_file_dir}/run_GWDIS_mclapply.r" \
      -iin="${data_file_dir}/chr1_251031_pruned.bed" \
      -iin="${data_file_dir}/chr1_251031_pruned.fam" \
      -iin="${txt_file_dir}/pheno_final_imputed_cox.csv" \

             export SNP_END=${end};

      --destination "${data_file_dir}" \
