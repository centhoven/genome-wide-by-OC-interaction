# estimate LD structure for only the SNPs in my sumstats (first run R script to create SNPfile and correct sumstats file)

# for first Chr

#6 munge GWAS summary statistics to ldsc-format

python munge_sumstats.py --sumstats sumstats_gwas_ldsc.txt --snp SNP --a1 A1 --a2 A2 --p P --signed-sumstats BETA,0 --N-con-col N_CON --N-cas-col N_CAS --merge-alleles custom_merge_alleles.txt --out depression

# Removed 71861 variants that were not SNPs or were strand-ambiguous.
# 403918 SNPs remain.

#7 estimate SNP h2

python ldsc.py --h2 depression.sumstats.gz --ref-ld-chr eur_ldscores. --w-ld-chr eur_ldscores. --pop-prev 0.1 --samp-prev 0.01 --out depression_h2

# Estimation of gene-by-OC-interaction component with PIGEON

# 8   Clone git environment (one time)

# 9  munge GWDIS summary statistics to ldsc-format

# 10 estimate de genexOC component

python PIGEON\pigeon.py --gxe-var --gxe-sumstats gwdis_munged.sumstats.gz --ref-ld-chr eur_ldscores. --w-ld-chr eur_ldscores. --pop-prev 0.1 --samp-prev 0.01 --out SNPxOC_depression_h2
